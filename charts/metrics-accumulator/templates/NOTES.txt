1. Get the application URL by running these commands:
{{- if .Values.ingress.enabled }}
{{- range $host := .Values.ingress.hosts }}
  {{- range .paths }}
  http{{ if $.Values.ingress.tls }}s{{ end }}://{{ $host.host }}{{ .path }}
  {{- end }}
{{- end }}
{{- else if contains "NodePort" .Values.service.type }}
  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "metrics-accumulator.fullname" . }})
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo "http://$NODE_IP:$NODE_PORT"
{{- else if contains "LoadBalancer" .Values.service.type }}
  NOTE: It may take a few minutes for the LoadBalancer IP to be available.
        You can watch the status by running 'kubectl get --namespace {{ .Release.Namespace }} svc -w {{ include "metrics-accumulator.fullname" . }}'
  export SERVICE_IP=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ include "metrics-accumulator.fullname" . }} --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
  echo "http://$SERVICE_IP:{{ .Values.service.port }}"
{{- else if contains "ClusterIP" .Values.service.type }}
  echo "Visit http://127.0.0.1:8080/swagger-ui/index.html to inspect the API"
  kubectl --namespace {{ .Release.Namespace }} port-forward svc/{{ include "metrics-accumulator.fullname" . }} 8080:{{ .Values.service.port }}
{{- end }}

2. Push Prometheus-formatted metrics to a metric group (`/update/{metricGroup}`):
   curl -X POST "http://127.0.0.1:8080/update/default" \
     -H "X-API-KEY: <your-metric-group-api-key>" \
     -H "Content-Type: text/plain" \
     --data-binary $'# TYPE job_runs_total counter\njob_runs_total{env="dev"} 1.0\n'

3. View accumulated metrics in Prometheus format (`/metrics/{metricGroup}`):
   curl "http://127.0.0.1:8080/metrics/default"

4. Inspect Prometheus HTTP service discovery targets (`/service-discovery`):
   curl "http://127.0.0.1:8080/service-discovery"

5. Configure Prometheus scraping:
   - Static target: {{ include "metrics-accumulator.fullname" . }}.{{ .Release.Namespace }}.svc.cluster.local:{{ .Values.service.port }} (metrics_path: /metrics/default)
   - HTTP SD endpoint: http://{{ include "metrics-accumulator.fullname" . }}.{{ .Release.Namespace }}.svc.cluster.local:{{ .Values.service.port }}/service-discovery
   - Ensure `honor_labels: true` is enabled in your Prometheus scrape configuration.
