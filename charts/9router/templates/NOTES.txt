9Router ({{ .Chart.AppVersion }}) has been deployed to namespace {{ .Release.Namespace }}.

1. Get the application URL by running these commands:
{{- if .Values.httpRoute.enabled }}
  HTTPRoute configured:
  {{- range .Values.httpRoute.hostnames }}
    https://{{ . }}
  {{- else }}
    kubectl get httproute {{ include "9router.fullname" . }} -n {{ .Release.Namespace }}
  {{- end }}
{{- else if .Values.ingress.enabled }}
  {{- range $host := .Values.ingress.hosts }}
  {{- range .paths }}
    http{{ if $.Values.ingress.tls }}s{{ end }}://{{ $host.host }}{{ .path }}
  {{- end }}
  {{- end }}
{{- else if contains "NodePort" .Values.service.type }}
  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "9router.fullname" . }})
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo http://$NODE_IP:$NODE_PORT
{{- else if contains "LoadBalancer" .Values.service.type }}
  NOTE: It may take a few minutes for the LoadBalancer IP to be available.
        Watch status with: kubectl get --namespace {{ .Release.Namespace }} svc -w {{ include "9router.fullname" . }}
  export SERVICE_IP=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ include "9router.fullname" . }} --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
  echo http://$SERVICE_IP:{{ .Values.service.port }}
{{- else }}
  kubectl --namespace {{ .Release.Namespace }} port-forward svc/{{ include "9router.fullname" . }} {{ .Values.service.port }}:{{ .Values.service.port }}
  echo "9Router health check: http://127.0.0.1:{{ .Values.service.port }}/api/health"
{{- end }}

2. Verify the release:
  kubectl get pods -n {{ .Release.Namespace }} -l app.kubernetes.io/instance={{ .Release.Name }}
  helm test {{ .Release.Name }} -n {{ .Release.Namespace }}
{{- if not .Values.persistence.enabled }}

Enable persistence.enabled=true in production to retain 9Router configuration and SQLite state across Pod restarts.
{{- end }}
