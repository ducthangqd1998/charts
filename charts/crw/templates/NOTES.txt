FastCRW (AI Web Scraper, Crawler & Search API) has been successfully deployed!

1. Get the FastCRW API service URL:
{{- if .Values.ingress.enabled }}
{{- range $host := .Values.ingress.hosts }}
  http{{ if $.Values.ingress.tls }}s{{ end }}://{{ $host.host }}
{{- end }}
{{- else if .Values.httpRoute.enabled }}
{{- range $host := .Values.httpRoute.hostnames }}
  https://{{ $host }}
{{- end }}
{{- else if contains "NodePort" .Values.service.type }}
  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "crw.fullname" . }})
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo "CRW URL: http://$NODE_IP:$NODE_PORT"
{{- else if contains "LoadBalancer" .Values.service.type }}
  NOTE: It may take a few minutes for the LoadBalancer IP to be available.
  export SERVICE_IP=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ include "crw.fullname" . }} --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
  echo "CRW URL: http://$SERVICE_IP:{{ .Values.service.port }}"
{{- else }}
  echo "Forwarding local port 3000 to CRW service..."
  kubectl --namespace {{ .Release.Namespace }} port-forward svc/{{ include "crw.fullname" . }} 3000:{{ .Values.service.port }}
{{- end }}
